#!/usr/bin/env python3
"""Generator key acak untuk BALEREOT COMMUNITY Toolbox.

=====================================================================
CARA PAKAI (ikuti urutan ini)
=====================================================================

1) BUAT KEY BARU  (jalankan di terminal / Termux / CMD)

   python3 KeyGenerator.py -n 5
       -> buat 5 key acak, BEBAS dipakai siapa saja

   python3 KeyGenerator.py -n 1 --username Budi123
       -> buat 1 key, HANYA bisa dipakai username Roblox "Budi123"

   python3 KeyGenerator.py -n 1 --username Budi123 --days 30
       -> sama seperti di atas, tapi kedaluwarsa 30 hari dari hari ini

   Hasilnya otomatis DITAMBAH ke file keys.txt (file lama tidak terhapus).

2) ISI keys.txt SEPERTI INI  (satu key per baris)

   # baris yang diawali tanda # diabaikan (boleh dipakai untuk catatan)
   BLR-E9Y8-DTG8-V5SU-TJQB|gahheyhdgdg            <- khusus user gahheyhdgdg
   BLR-GJKA-UXKJ-7ZED-Z5HE|Budi123|2026-12-31     <- khusus Budi123, sampai 31 Des 2026
   BLR-V26F-V9E3-MKTP-SS3D|*                      <- bebas, siapa saja boleh pakai

   Format:  KEY | USERNAME (atau *) | TANGGAL (boleh dikosongkan)

3) UPLOAD keys.txt KE GITHUB / PASTEBIN
   Ambil link RAW-nya, tempel SEKALI di MyToolbox.lua pada KEY_LIST_URL.
   Setelah itu skrip selalu membaca daftar terbaru dari link tersebut.

4) MENAMBAH KEY LAGI
   Jalankan lagi langkah 1, lalu update isi file di GitHub dengan keys.txt terbaru.
   MyToolbox.lua TIDAK perlu diubah.

5) MENGHAPUS / MENCABUT KEY
   Buka file daftar key (keys.txt di GitHub), HAPUS BARIS key tersebut, simpan.
   Contoh: mau cabut key Budi123 -> hapus baris ini:
       BLR-GJKA-UXKJ-7ZED-Z5HE|Budi123|2026-12-31
   Key itu langsung tidak berlaku lagi saat orang membuka skrip berikutnya.

6) GANTI USERNAME PEMILIK KEY
   Hapus baris lama, lalu buat key baru dengan --username yang baru.
"""
import argparse
import secrets
from datetime import date, timedelta

ALPHABET = "ABCDEFGHJKLMNPQRSTUVWXYZ23456789"  # tanpa huruf/angka yang mirip (O, 0, I, 1)


def make_key(prefix="BLR", groups=4, size=4):
    parts = ["".join(secrets.choice(ALPHABET) for _ in range(size)) for _ in range(groups)]
    return "-".join([prefix] + parts)


def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("-n", "--count", type=int, default=10)
    ap.add_argument("--days", type=int, default=0, help="masa berlaku (hari), 0 = tanpa batas")
    ap.add_argument("--username", default="*", help="Username Roblox untuk mengunci key, * = bebas")
    ap.add_argument("-o", "--out", default="keys.txt")
    args = ap.parse_args()

    expiry = (date.today() + timedelta(days=args.days)).isoformat() if args.days > 0 else ""
    lines = []
    for _ in range(args.count):
        line = f"{make_key()}|{args.username}"
        if expiry:
            line += f"|{expiry}"
        lines.append(line)

    with open(args.out, "a", encoding="utf-8") as f:
        f.write("\n".join(lines) + "\n")
    print("\n".join(lines))
    print(f"\n{len(lines)} key ditambahkan ke {args.out}")


if __name__ == "__main__":
    main()
